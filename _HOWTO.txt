2️⃣ How UE4SS mods like CNS do it
	CNS mod does not create new classes. What it does:
	Loads SkeletalMesh and Material assets from .pak file
	Adds them to existing arrays / DataTables (like SuitList: TArray<FSuitData>)
	Hooks the UI menu (WBP_NanosuitMenu) to read the new entries
	Swaps meshes on the player at runtime via the game’s existing functions
	New Model + Material → FSuitData → TArray → UI Menu → SwapMesh(Player)

		local suitData = FSuitData()
		suitData.Name = "MyNewSuit"
		suitData.Mesh = LoadObject("/Game/MyMod/MySuitMesh.MySuitMesh")
		suitData.Material = LoadObject("/Game/MyMod/MySuitMat.MySuitMat")

		-- Add to suit list
		PlayerSuitList:Add(suitData)

	c code
		// memread.c
		#include <lua.h>
		#include <lauxlib.h>
		#include <stdint.h>

		static int l_read_memory(lua_State* L)
		{
			uintptr_t addr = (uintptr_t)lua_tointeger(L, 1);
			size_t size    = (size_t)lua_tointeger(L, 2);

			lua_pushlstring(L, (const char*)addr, size);
			return 1;
		}

		static const luaL_Reg memread_lib[] = {
			{"read", l_read_memory},
			{NULL, NULL}
		};

		__declspec(dllexport)
		int luaopen_memread(lua_State* L)
		{
			luaL_newlib(L, memread_lib);
			return 1;
		}
